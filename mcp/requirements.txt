# Dependencies for the aigents MCP server (see model_mcp_server.py).
#
# Pinned lower bounds only. The MCP SDK is a fast-moving project and made a
# breaking rename between 1.x and 2.x (FastMCP -> MCPServer), so this is
# validated against mcp 2.x. Bumping past a major version needs a re-check of
# mcp/server/mcpserver imports.
#
# The server deliberately avoids the openai/httpx client SDKs: it talks to
# llama.cpp with urllib from the stdlib, so `mcp` is the only real dependency.
mcp>=2.3,<3

# ASGI server + framework for the streamable-http transport. Both come from the
# `mcp[cli]` extra but are listed explicitly to make the requirement obvious.
starlette>=0.40
uvicorn>=0.30
